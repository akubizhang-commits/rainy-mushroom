// Copied without behavioral changes from verify-tower-journey.mjs.

export function steer(run, pursue = false) {
  const p = run.state.player, theta = run.state.time * .24
  let dx = 33 + 490 * Math.cos(theta) - p.x, dy = -60 + 490 * Math.sin(theta) - p.y
  const d = Math.max(1, Math.hypot(dx, dy)); dx /= d; dy /= d
  const near = [...run.state.enemies].sort((a, b) =>
    Math.hypot(a.x - p.x, a.y - p.y) - Math.hypot(b.x - p.x, b.y - p.y))[0]
  if (pursue && near) {
    const distance = Math.hypot(near.x - p.x, near.y - p.y)
    if (distance > 250) { dx = (near.x - p.x) / distance; dy = (near.y - p.y) / distance }
  }
  for (const e of run.state.enemies) {
    const distance = Math.hypot(p.x - e.x, p.y - e.y)
    if (distance < 135) {
      dx += (p.x - e.x) / Math.max(1, distance) * (135 - distance) / 45
      dy += (p.y - e.y) / Math.max(1, distance) * (135 - distance) / 45
    }
  }
  if (pursue && (run.state.zones.some(z=>z.hostile) || run.state.projectiles.some(b=>b.hostile))) {
    const norm=Math.max(1,Math.hypot(dx,dy)), bx=dx/norm, by=dy/norm
    let best=-Infinity, chosen={x:bx,y:by}
    for(let i=0;i<24;i++){
      const x=Math.cos(i*Math.PI/12),y=Math.sin(i*Math.PI/12)
      let score=x*bx+y*by
      for(const lead of [.25,.5,.85]){
        const q={x:p.x+x*p.moveSpeed*lead,y:p.y+y*p.moveSpeed*lead}
        if(q.x < -860 || q.x > 925 || q.y < -1080 || q.y > 960)score-=15
        for(const z of run.state.zones.filter(z=>z.hostile)){
          let danger=false
          if(z.shape==='capsule'){
            const ax=z.x,ay=z.y,ex=z.toX-ax,ey=z.toY-ay
            const t=Math.max(0,Math.min(1,((q.x-ax)*ex+(q.y-ay)*ey)/(ex*ex+ey*ey||1)))
            danger=Math.hypot(q.x-ax-t*ex,q.y-ay-t*ey)<(z.width||24)/2+28
          }else if(z.shape==='sector'){
            const ex=q.x-z.x,ey=q.y-z.y,d=Math.hypot(ex,ey)
            danger=d<(Math.hypot(z.toX-z.x,z.toY-z.y)||z.radius)+22
              && (ex*z.faceX+ey*z.faceY)/(d||1)>Math.cos((z.angle||90)*Math.PI/360)
          }else danger=Math.hypot(q.x-z.x,q.y-z.y)<z.radius+28
          if(danger)score-=8
        }
        for(const b of run.state.projectiles.filter(b=>b.hostile)){
          if(Math.hypot(q.x-b.x-(b.vx||0)*lead,q.y-b.y-(b.vy||0)*lead)<(b.radius||8)+25)score-=8
        }
      }
      if(score>best){best=score;chosen={x,y}}
    }
    dx=chosen.x;dy=chosen.y
  }
  run.setInput(dx, dy)
}

// Read-only observation of unique effect ids: actual effective hit damage, not raw skill coefficients.
// The observer never patches Run internals, HP, RNG, enemies, cooldowns or rewards.
export function play(run, { maxSeconds = 1800, pursue = true, checkpointEvery = 80,
  onCheckpoint, onWave, onStep } = {}) {
  const metrics = { waves: [], autoSkills: {}, firstCast: {}, firstAction: {}, effectiveDamage: 0, hpDamage: 0,
    hitEvents: 0, peakEnemies: 0, timeout: false }
  let nextCheckpoint = checkpointEvery, previousWaves = run.state.clearedWaves.length
  let seenEffects = new Set(), seenCasts = new Set()
  const spawnedHp = new Map()
  function observe() {
    const s = run.state
    metrics.peakEnemies = Math.max(metrics.peakEnemies, s.enemies.length)
    for(const enemy of s.enemies)if(!spawnedHp.has(enemy.id))spawnedHp.set(enemy.id,enemy.maxHp)
    for (const id of Object.keys(s.loadout.active)) metrics.autoSkills[id] ??= rounded(s.time)
    const current = new Set()
    for (const fx of s.effects) {
      current.add(fx.id)
      if (seenEffects.has(fx.id)) continue
      const skill = {orbit:'sj_wanjian',wuwo:'sj_wanjian',arc:'sj_arc',line:'sj_line',tiandi:'sj_line'}[fx.kind]
      if (skill) metrics.firstAction[skill] ??= rounded(s.time)
      if (fx.kind === 'hit') { metrics.effectiveDamage += fx.damage || 0; metrics.hitEvents++ }
      if (fx.kind === 'player_hit') metrics.hpDamage += fx.damage || 0
    }
    seenEffects = current
    for(const shot of s.projectiles)if(!shot.hostile&&shot.source)metrics.firstAction[shot.source]??=rounded(s.time)
    for(const zone of s.zones){const id={shadow:'sj_shadow',formation:'sj_formation',sword_rain:'evo_wanjian'}[zone.kind];if(!zone.hostile&&id)metrics.firstAction[id]??=rounded(s.time)}
    const casts = new Set()
    for (const cast of s.casts) {
      const key = `${cast.skill}:${cast.releaseAt}`; casts.add(key)
      if (!seenCasts.has(key)) metrics.firstCast[cast.skill] ??= rounded(s.time)
    }
    seenCasts = casts
  }
  observe()
  // Advance one actual engine tick to avoid losing short-lived hit effects at wave transitions.
  while (run.state.phase === 'playing' && run.state.time < maxSeconds) {
    if (pursue) {
      const p = run.state.player
      run.setViewport({ xmin:p.x-352, xmax:p.x+352, ymin:p.y-506, ymax:p.y+511 })
    }
    steer(run, pursue)
    run.advance(1 / 60)
    observe()
    onStep?.(run, metrics)
    if (run.state.clearedWaves.length > previousWaves) {
      const s = run.state
      for (const wave of s.clearedWaves.slice(previousWaves)) {
        const record = { wave, seconds:rounded(s.time), hp:rounded(s.player.hp), xp:s.player.xp,
          autoSkills:clone(s.loadout.active), tutorialWaves:s.loadout.tutorialWaves ?? null }
        metrics.waves.push(record); onWave?.(record, run)
      }
      previousWaves = s.clearedWaves.length
    }
    if (onCheckpoint && run.state.phase === 'playing' && run.state.time >= nextCheckpoint) {
      run.pause()
      run = onCheckpoint(run) || run
      assert.equal(run.state.phase, 'paused')
      run.resume(); nextCheckpoint += checkpointEvery
    }
  }
  if (run.state.phase === 'playing') { metrics.timeout = true; run.abandon() }
  const killedHp = run.state.killLedger.reduce((sum,k)=>sum+(spawnedHp.get(k.id)||0),0)
  const survivingDamage = run.state.enemies.reduce((sum,e)=>sum+e.maxHp-e.hp,0)
  metrics.damageAudit = { spawnedCount:spawnedHp.size,killedHp:rounded(killedHp),survivingDamage:rounded(survivingDamage),
    expectedEffectiveDamage:rounded(killedHp+survivingDamage),difference:rounded(metrics.effectiveDamage-killedHp-survivingDamage),
    exactWithinTolerance:Math.abs(metrics.effectiveDamage-killedHp-survivingDamage)<1e-5 }
  metrics.effectiveDamage = rounded(metrics.effectiveDamage)
  metrics.hpDamage = rounded(metrics.hpDamage)
  metrics.effectiveDps = rounded(metrics.effectiveDamage / Math.max(1,run.state.time))
  return { run, metrics }
}

async function runtime() {
  const growthPath = resolve(ROOT,'js/xianxia/TowerGrowth.js')
  if (!existsSync(growthPath)) throw Error('DEPENDENCY_NOT_READY: TowerGrowth.js has not landed')
  const [growth, campaign, combat, profile, content, save] = await Promise.all([
    import(pathToFileURL(growthPath)), import('../js/xianxia/TowerCampaign.js'),
    import('../js/xianxia/TowerRun.js'), import('../js/xianxia/TowerProfile.js'),
    import('../js/xianxia/TowerContent.js'), import('../js/survivor/SurvivorSave.js'),
  ])
  for (const name of ['growthSummary','realmPoints','skillRequirement']) assert.equal(typeof growth[name],'function',name)
  const probe = new campaign.TowerCampaignStore({getItem:()=>null,setItem:()=>{}},0)
  assert.equal(campaign.combatLoadout(probe.state).growthVersion,1,'DEPENDENCY_NOT_READY: Campaign growth loadout')
  assert.equal(typeof combat.TowerRun.prototype.setGrowthLoadout,'function','DEPENDENCY_NOT_READY: TowerRun growth API')
  for (const f of content.FLOORS) assert.ok(f.waves.length >= 20,`floor ${f.floor} must have >=20 waves`)
  return {...growth,...campaign,...combat,...profile,checksum:save.checksum,FLOORS:content.FLOORS}
}
function context(rt, label, seed, fixture = false) {
  const memory = new Map(), storage = {getItem:k=>memory.get(k)??null,setItem:(k,v)=>memory.set(k,v)}
  const ctx = {label, seed, fixture, now:0, storage, memory, transactions:[], denied:{}, store:new rt.TowerCampaignStore(storage,0)}
  ctx.summary = () => ({xp:ctx.store.state.xp,hp:ctx.store.state.hp,cores:ctx.store.state.cores,
    contribution:ctx.store.state.contribution,points:rt.growthSummary(ctx.store.state),
    pills:clone(ctx.store.state.pills),skills:clone(ctx.store.state.skills),
    evolutionRanks:clone(ctx.store.state.evolutionRanks),books:clone(ctx.store.state.books),
    weapon:clone(ctx.store.state.weapons.eq_sword_suwen),clearedFloor:ctx.store.state.clearedFloor})
  ctx.act = action => {
    const before=ctx.summary(), result=ctx.store.act(action,ctx.now)
    if (!result.ok) {
      const key=`${action.type}:${action.id||''}:${result.reason}`
      ctx.denied[key]=(ctx.denied[key]||0)+1
      return result
    }
    const after=ctx.summary()
    const entry={atMs:ctx.now,action:{...action},result:clone(result),before,after}
    if (entry.action.snapshot) entry.action.snapshot={sha256:digest(action.snapshot),floor:action.snapshot.floor,
      seconds:rounded(action.snapshot.time),waves:action.snapshot.clearedWaves.length,phase:action.snapshot.phase}
    if (entry.result.snapshot) entry.result.snapshot={sha256:digest(result.snapshot)}
    ctx.transactions.push(entry)
    if (action.type==='learnSkill') {
      assert.ok(after.points.spent > before.points.spent,'learnSkill must actually spend SP')
      assert.ok(after.cores<=before.cores,'learnSkill cannot mint cores')
      entry.pointCost=after.points.spent-before.points.spent;entry.bookCost=before.cores-after.cores
    }
    if(action.type==='researchPoint') {
      assert.equal(before.cores-after.cores,[80,160,240][before.points.research],'actual research debit')
      assert.equal(after.points.total-before.points.total,1,'research awards exactly one point')
    }
    assert.equal(after.points.spent+after.points.available,after.points.total)
    assert.ok(after.points.total<=36)
    return result
  }
  ctx.reload=()=>{
    const before=clone(ctx.store.state)
    ctx.store=new rt.TowerCampaignStore(storage,ctx.now)
    assert.equal(ctx.store.readOnly,false,ctx.store.error)
    assert.deepEqual(ctx.store.state,before,'persisted state roundtrip')
  }
  assert.ok(ctx.act({type:'sync'}).ok)
  return ctx
}
function purchaseRoute(rt,ctx,route,inTower=false,priorities=route.targets) {
  const allowed=route.budget===30 ? Math.min(30,rt.realmPoints(ctx.store.state.xp)) : 36
  let previousSpent = -1
  while(previousSpent !== rt.growthSummary(ctx.store.state).spent) {
    previousSpent=rt.growthSummary(ctx.store.state).spent
  for (const [id,target] of priorities) {
    if (route.budget===30 && id==='sj_shadow') continue
    while ((ctx.store.state.skills[id]||ctx.store.state.evolutionRanks[id]||0)<target) {
      const req=rt.skillRequirement(ctx.store.state,id,{inTower})
      if (!req.ok || rt.growthSummary(ctx.store.state).spent+req.cost>allowed) break
      const before=ctx.summary(), result=ctx.act({type:'learnSkill',id})
      if (!result.ok) break
      assert.equal(ctx.store.state.cores,before.cores-(req.bookCost||0),'book fee really debited')
      assert.equal(rt.growthSummary(ctx.store.state).spent-before.points.spent,req.cost,'SP really debited')
    }
  }
  }
}
function prepareCamp(rt,ctx,route) {
  if (route.budget===36) for(let i=0;i<3;i++) if(!ctx.act({type:'researchPoint'}).ok) break
  // Medicine is reserved before refinement. No artificial waiting or direct HP writes.
  for(let i=0;i<2&&ctx.store.state.hp<rt.cultivationAt(ctx.store.state.xp).maxHp-1e-6;i++) {
    if(!ctx.store.state.pills.huichun && !ctx.act({type:'buyPill',id:'huichun',count:1}).ok) break
    if(!ctx.act({type:'usePill',id:'huichun'}).ok) break
  }
  if(ctx.store.state.contribution>=150)ctx.act({type:'refineWeapon',id:'eq_sword_suwen'})
  // At most one actually paid XP pill per visit; contribution stays available for two heals.
  if(ctx.store.state.xp<rt.TOWER_XP_CAP&&ctx.store.state.contribution>=70) {
    if(ctx.act({type:'buyPill',id:'peiyuan',count:1}).ok)assert.ok(ctx.act({type:'usePill',id:'peiyuan'}).ok)
  }
  purchaseRoute(rt,ctx,route,false,NATURAL_PRIORITY)
  ctx.reload()
}
function simulateAttempt(rt,ctx,floor,seed,options,{learn=true,route}={}) {
  const before=ctx.summary(), started=ctx.act({type:'startFloor',floor,seed})
  assert.ok(started.ok,`startFloor:${started.reason}`)
  assert.equal(started.snapshot.loadout.growthVersion,1)
  let checkpointTime=0
  const played=play(rt.TowerRun.restore(started.snapshot),{maxSeconds:options.maxSeconds,
    onWave:(wave,run)=>{
      if(floor===1&&before.xp===0){
        if(wave.wave===1)assert.deepEqual(Object.keys(wave.autoSkills).sort(),['sj_arc','sj_yujian'])
        if(wave.wave===3)assert.deepEqual(Object.keys(wave.autoSkills).sort(),['sj_arc','sj_wanjian','sj_yujian'])
      }
    },
    onCheckpoint:run=>{
      const snapshot=run.snapshot()
      assert.ok(rt.validateTowerSnapshot(snapshot),'paused snapshot validation')
      ctx.now+=Math.round((snapshot.time-checkpointTime)*1000);checkpointTime=snapshot.time
      const result=ctx.act({type:'checkpoint',snapshot})
      assert.ok(result.ok,`checkpoint:${result.reason}`)
      ctx.reload()
      if(learn)purchaseRoute(rt,ctx,route,true,NATURAL_PRIORITY)
      const saved=ctx.store.state.activeRun.snapshot
      assert.equal(saved.player.hp,snapshot.player.hp,'paused learning preserves HP')
      assert.equal(saved.time,snapshot.time,'paused learning preserves time')
      assert.deepEqual(saved.enemies,snapshot.enemies,'paused learning preserves enemies')
      assert.deepEqual(saved.killLedger,snapshot.killLedger,'paused learning preserves kills')
      return rt.TowerRun.restore(saved)
    }})
  const snapshot=played.run.snapshot()
  assert.equal(played.metrics.damageAudit.exactWithinTolerance,true,'hit effects reconcile with real killed and surviving HP')
  assert.ok(rt.validateTowerSnapshot(snapshot),'terminal snapshot validation')
  ctx.now+=Math.round((snapshot.time-checkpointTime)*1000)
  const finished=ctx.act({type:'finishFloor',snapshot})
  assert.ok(finished.ok,`finishFloor:${finished.reason}`)
  assert.equal(ctx.act({type:'finishFloor',snapshot}).ok,false,'duplicate settlement rejected')
  ctx.reload()
  if(snapshot.phase==='won')assert.equal(snapshot.clearedWaves.length,rt.FLOORS[floor-1].waves.length)
  if(snapshot.phase==='lost')assert.equal(ctx.store.state.hp,1)
  return {floor,seed,outcome:snapshot.phase,seconds:rounded(snapshot.time),clearedWaves:snapshot.clearedWaves.length,
    kills:snapshot.killLedger.length,start:before,end:ctx.summary(),earned:finished.earned,
    loadout:clone(snapshot.loadout),metrics:played.metrics,snapshotHash:digest(snapshot)}
}
async function naturalJourney(rt,routeKey,seed,options,progress) {
  const route=ROUTES[routeKey],ctx=context(rt,`natural-${routeKey}-${seed}`,seed)
  const record={kind:'natural: starter grants + real kill/wave awards + paid camp actions',route:routeKey,seed,
    attempts:[],initial:ctx.summary(),transactions:ctx.transactions,denied:ctx.denied}
  // Persist partial progress even if a later runtime integration assertion fails.
  progress(record)
  try {
    for(let floor=1;floor<=9;floor++) {
      let won=false
      const limit=floor===9?options.maxAttempts:Math.min(12,options.maxAttempts)
      for(let attempt=1;attempt<=limit&&!won;attempt++) {
        if(floor!==1||attempt!==1)prepareCamp(rt,ctx,route)
        const runSeed=(seed+floor*1009+attempt*9176)>>>0
        const result=simulateAttempt(rt,ctx,floor,runSeed,options,{route})
        result.attempt=attempt;record.attempts.push(result)
        won=result.outcome==='won'
        console.log(`${ctx.label} f${floor} a${attempt}: ${result.outcome}, ${result.seconds}s, ${result.clearedWaves}/20, XP ${result.end.xp}, SP ${result.end.points.spent}/${result.end.points.total}`)
        record.final=ctx.summary();progress()
      }
      if(!won){record.stoppedAtFloor=floor;break}
    }
    // Return to sect after the final attempt; report post-clear unlocks separately from fight strength.
    prepareCamp(rt,ctx,route)
    record.final=ctx.summary();record.reachedNine=ctx.store.state.clearedFloor===9
    record.totalCombatSeconds=rounded(record.attempts.reduce((sum,a)=>sum+a.seconds,0))
    record.transactionHash=digest(ctx.transactions)
    return record
  }catch(error){record.error=error.stack;record.final=ctx.summary();throw error}
  finally{progress()}
}
function fixtureContext(rt,routeKey,seed) {
  const route=ROUTES[routeKey],ctx=context(rt,`fixture-${routeKey}-${seed}`,seed,true)
  // Explicit capacity/combat fixture only. Seed a valid schema record, never mutate Run HP/enemies.
  // Artificial sources are recorded separately; all books, research and skill ranks are purchased via act.
  const envelope=JSON.parse(ctx.memory.get(rt.TOWER_CAMPAIGN_KEY))
  const state=JSON.parse(envelope.data)
  Object.assign(state,{xp:rt.TOWER_XP_CAP,hp:rt.cultivationAt(rt.TOWER_XP_CAP).maxHp,clearedFloor:9,
    cores:2000,contribution:300,tutorialWaves:3,researchPurchases:0,
    skills:{sj_yujian:1,sj_arc:1,sj_wanjian:1},evolutions:[],evolutionRanks:{},books:[]})
  // Same starter weapon, upgraded through four real refinement transactions, for every route.
  envelope.data=JSON.stringify(state);envelope.checksum=rt.checksum(envelope.data)
  ctx.memory.set(rt.TOWER_CAMPAIGN_KEY,JSON.stringify(envelope))
  ctx.store=new rt.TowerCampaignStore(ctx.storage,0)
  assert.equal(ctx.store.readOnly,false,`fixture envelope:${ctx.store.error}`)
  ctx.fixtureSources={xp:rt.TOWER_XP_CAP,clearedFloor:9,tutorialWaves:3,cores:2000,contribution:300,
    hp:state.hp,note:'Artificial capacity/combat control, NOT natural progression. 30 route leaves automatic first-clear points unspent.'}
  ctx.fixtureBefore=ctx.summary()
  if(route.budget===36)for(let i=0;i<3;i++)assert.ok(ctx.act({type:'researchPoint'}).ok)
  for(let i=0;i<4;i++)assert.ok(ctx.act({type:'refineWeapon',id:'eq_sword_suwen'}).ok)
  purchaseRoute(rt,ctx,route)
  assert.equal(rt.growthSummary(ctx.store.state).spent,route.budget,`${routeKey} fixture budget`)
  ctx.reload()
  return ctx
}

function optionsFrom(args) {
  const options={seeds:[1,7,43],maxAttempts:40,maxSeconds:1800,mode:'all',suffix:'',repeatCheck:false}
  for(const arg of args){
    if(arg==='--repeat-check'){options.repeatCheck=true;continue}
    const match=/^--([a-z-]+)=(.+)$/.exec(arg);assert.ok(match,`unknown argument ${arg}`)
    const [,key,value]=match
    if(key==='seeds')options.seeds=value.split(',').map(Number)
    else if(key==='max-attempts')options.maxAttempts=Number(value)
    else if(key==='max-seconds')options.maxSeconds=Number(value)
    else if(key==='mode')options.mode=value
    else if(key==='suffix')options.suffix=value
    else throw Error(`unknown option ${key}`)
  }
  assert.ok(options.seeds.length>0&&options.seeds.every(s=>Number.isInteger(s)&&s>=0&&s<=0xffffffff))
  assert.ok(Number.isInteger(options.maxAttempts)&&options.maxAttempts>=1&&options.maxAttempts<=40)
  assert.ok(Number.isFinite(options.maxSeconds)&&options.maxSeconds>=1&&options.maxSeconds<=7200)
  assert.ok(['all','natural','fixtures'].includes(options.mode))
  assert.ok(/^(-[a-zA-Z0-9-]+)?$/.test(options.suffix))
  return options
}
export async function main(args=process.argv.slice(2)) {
  const options=optionsFrom(args),sourceFiles=['TowerGrowth','TowerCampaign','TowerRun','TowerProfile','TowerContent']
  const hashes=()=>Object.fromEntries(sourceFiles.map(f=>{const p=resolve(ROOT,`js/xianxia/${f}.js`);return [f,existsSync(p)?digest(readFileSync(p,'utf8')):null]}))
  const report={kind:'unmodified growth combat bot QA; NOT human balance acceptance',options,nodeVersion:process.version,
    scriptHash:digest(readFileSync(fileURLToPath(import.meta.url),'utf8')),
    legacySteerSourceHash:digest(readFileSync(resolve(ROOT,'scripts/verify-tower-journey.mjs'),'utf8')), 
    sourceHashes:hashes(),natural:[],fixtures:[],status:'running'}
  const flush=()=>{persist(report,options.suffix);writeMarkdown(report,options.suffix)}
  flush()
  try {
    const rt=await runtime()
    if(options.mode!=='fixtures')for(const route of ['pursuit36','pursuit30'])for(const seed of options.seeds)
      await naturalJourney(rt,route,seed,options,record=>{if(record)report.natural.push(record);flush()})
    if(options.mode!=='natural')for(const route of Object.keys(ROUTES))for(const seed of options.seeds){
      const ctx=fixtureContext(rt,route,seed),combat=simulateAttempt(rt,ctx,9,seed,options,{learn:false})
      const entry={kind:'CAPACITY/COMBAT FIXTURE — NOT natural progression',route,seed,
        sources:ctx.fixtureSources,beforePurchases:ctx.fixtureBefore,combat,transactions:ctx.transactions}
      report.fixtures.push(entry)
      console.log(`fixture ${route}/${seed}: ${combat.outcome}, ${combat.seconds}s, ${combat.clearedWaves}/20, DPS ${combat.metrics.effectiveDps}`)
      if(options.repeatCheck&&!report.reproducibility){
        const twin=fixtureContext(rt,route,seed),again=simulateAttempt(rt,twin,9,seed,options,{learn:false})
        assert.equal(again.snapshotHash,combat.snapshotHash,'seed replay snapshot hash')
        assert.equal(digest(twin.transactions),digest(ctx.transactions),'seed replay transaction hash')
        assert.deepEqual(again.metrics,combat.metrics,'seed replay metrics')
        report.reproducibility={route,seed,passed:true,snapshotHash:combat.snapshotHash,transactionHash:digest(ctx.transactions)}
      }
      flush()
    }
    report.sourceHashesAfter=hashes();report.sourceStable=digest(report.sourceHashes)===digest(report.sourceHashesAfter)
    const pressure=report.natural.some(j=>!j.reachedNine)||report.fixtures.some(f=>f.combat.outcome!=='won')
    report.status=!report.sourceStable?'source-changed-rerun-required':pressure?'completed-with-combat-failures':'completed'
    if(!report.sourceStable)process.exitCode=3
    // Observed combat loss is a reportable result, not a script assertion failure or manufactured pass.
  }catch(error){report.status=String(error).includes('DEPENDENCY_NOT_READY')?'dependency-not-ready':'integration-failure';report.error=error.stack;process.exitCode=2;console.error(error.stack)}
  finally{flush()}
  return report
}
